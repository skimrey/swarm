```python
def parse_duration(s) -> int:
    if not isinstance(s, str) or s == "":
        raise ValueError("invalid duration")

    values = {"d": 86400, "h": 3600, "m": 60, "s": 1}
    order = {"d": 0, "h": 1, "m": 2, "s": 3}

    total = 0
    last_unit_index = -1
    i = 0
    n = len(s)

    while i < n:
        start = i

        while i < n and "0" <= s[i] <= "9":
            i += 1

        if i == start:
            raise ValueError("invalid duration")

        if i == n:
            raise ValueError("invalid duration")

        unit = s[i]
        if unit not in values:
            raise ValueError("invalid duration")

        unit_index = order[unit]
        if unit_index <= last_unit_index:
            raise ValueError("invalid duration")

        total += int(s[start:i]) * values[unit]
        last_unit_index = unit_index
        i += 1

    return total
```
